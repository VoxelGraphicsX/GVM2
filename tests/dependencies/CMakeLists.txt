cmake_minimum_required(VERSION 3.22)
project(GVMDependencyConsumer LANGUAGES C CXX)

# Dependency discovery must leave the caller's policy override unchanged.
set(_gvm_caller_policy_defined OFF)
if(DEFINED CMAKE_POLICY_VERSION_MINIMUM)
    set(_gvm_caller_policy_defined ON)
    set(_gvm_caller_policy_version "${CMAKE_POLICY_VERSION_MINIMUM}")
endif()
get_property(_gvm_caller_policy_cached CACHE CMAKE_POLICY_VERSION_MINIMUM PROPERTY TYPE SET)

if(NOT TARGET GVM::GVM)
    if(GVM_TEST_PREEXISTING_TARGET)
        add_library(${GVM_TEST_PREEXISTING_TARGET} INTERFACE IMPORTED GLOBAL)
    endif()
    if(GVM_TEST_SOURCE_DIR)
        add_subdirectory("${GVM_TEST_SOURCE_DIR}" gvm)
    else()
        find_package(GVM CONFIG REQUIRED)
        # Repeated discovery must reuse GVM's dependency provider.
        find_package(GVM CONFIG REQUIRED)
    endif()
endif()

if(_gvm_caller_policy_defined)
    if(NOT DEFINED CMAKE_POLICY_VERSION_MINIMUM OR
       NOT CMAKE_POLICY_VERSION_MINIMUM STREQUAL _gvm_caller_policy_version)
        message(FATAL_ERROR "GVM changed the caller's CMake policy override.")
    endif()
elseif(DEFINED CMAKE_POLICY_VERSION_MINIMUM)
    message(FATAL_ERROR "GVM leaked a CMake policy override into the caller.")
endif()
get_property(_gvm_policy_cached CACHE CMAKE_POLICY_VERSION_MINIMUM PROPERTY TYPE SET)
if(NOT _gvm_policy_cached STREQUAL _gvm_caller_policy_cached)
    message(FATAL_ERROR "GVM changed the cache state of the caller's CMake policy override.")
endif()
if(CMAKE_VERSION VERSION_GREATER_EQUAL 4.0)
    get_target_property(_gvm_eabase_source EABase SOURCE_DIR)
    get_directory_property(_gvm_eabase_policy DIRECTORY "${_gvm_eabase_source}"
        DEFINITION CMAKE_POLICY_VERSION_MINIMUM)
    if(_gvm_eabase_policy VERSION_LESS 3.5 OR
       (_gvm_caller_policy_defined AND _gvm_eabase_policy VERSION_LESS _gvm_caller_policy_version))
        message(FATAL_ERROR "EABase did not retain the required CMake policy minimum.")
    endif()
    foreach(_gvm_dependency EASTL glm-header-only)
        get_target_property(_gvm_dependency_source ${_gvm_dependency} SOURCE_DIR)
        get_directory_property(_gvm_dependency_policy DIRECTORY "${_gvm_dependency_source}"
            DEFINITION CMAKE_POLICY_VERSION_MINIMUM)
        if(NOT "${_gvm_dependency_policy}" STREQUAL "${_gvm_caller_policy_version}")
            message(FATAL_ERROR "EABase's policy override leaked into ${_gvm_dependency}.")
        endif()
    endforeach()
endif()

enable_testing()
add_library(gvm_dependency_boundary STATIC boundary.cpp)
target_link_libraries(gvm_dependency_boundary PUBLIC GVM::GVM)
target_compile_definitions(gvm_dependency_boundary PRIVATE NDEBUG)

foreach(_variant normal debug)
    add_executable(gvm_dependency_${_variant} main.cpp glm_first.cpp gvm_first.cpp)
    target_link_libraries(gvm_dependency_${_variant} PRIVATE gvm_dependency_boundary GVM::GLM GVM::EASTL)
    if(_variant STREQUAL "debug")
        target_compile_definitions(gvm_dependency_${_variant} PRIVATE _DEBUG=1 EA_DEBUG=1)
    endif()
    add_test(NAME dependency_${_variant} COMMAND gvm_dependency_${_variant})
endforeach()

add_executable(gvm_glm_standalone glm_standalone.cpp)
target_link_libraries(gvm_glm_standalone PRIVATE GVM::GLM)
add_test(NAME dependency_glm_standalone COMMAND gvm_glm_standalone)
add_executable(gvm_eastl_standalone eastl_standalone.cpp)
target_link_libraries(gvm_eastl_standalone PRIVATE GVM::EASTL)
add_test(NAME dependency_eastl_standalone COMMAND gvm_eastl_standalone)
